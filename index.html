document.addEventListener("DOMContentLoaded", function () { // 1. Parse the URL parameters (e.g., ?view=rider or ?view=driver) const urlParams = new URLSearchParams(window.location.search); const viewMode = urlParams.get('view'); // 2. Select your container elements (update these IDs to match your HTML elements) const riderSection = document.getElementById('rider-section'); const driverSection = document.getElementById('driver-section'); const payoutsSection = document.getElementById('payouts-section'); const topNavTabs = document.getElementById('top-nav-tabs'); // Optional: top navigation bar containing the tabs // 3. Enforce strict role and interface isolation if (viewMode === 'rider') { // RIDER APP VIEW: Show only rider booking, hide all driver/compliance/payout panels if (riderSection) riderSection.style.display = 'block'; if (driverSection) driverSection.style.display = 'none'; if (payoutsSection) payoutsSection.style.display = 'none'; if (topNavTabs) topNavTabs.style.display = 'none'; // Hides the tab switcher completely console.log("RideFlow Security: Loaded in secure RIDER mode."); } else if (viewMode === 'driver') { // DRIVER APP VIEW: Show driver compliance and payouts, hide rider booking if (riderSection) riderSection.style.display = 'none'; if (driverSection) driverSection.style.display = 'block'; if (payoutsSection) payoutsSection.style.display = 'block'; if (topNavTabs) topNavTabs.style.display = 'none'; // Hides the tab switcher completely console.log("RideFlow Security: Loaded in secure DRIVER mode."); } else { // DEFAULT/ADMIN VIEW: If no view parameter is specified, show everything (for local testing or web admin) console.log("RideFlow Security: Loaded in default/admin mode."); } });
